cmake_minimum_required(VERSION 3.16)
project(DackDSA LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# main.cpp includes the modules; compile only this translation unit.
add_executable(dsa_demo main.cpp)
add_executable(dsa_server server.cpp)
find_package(Threads REQUIRED)
target_link_libraries(dsa_server PRIVATE Threads::Threads)
if(WIN32)
  target_link_libraries(dsa_server PRIVATE ws2_32)
endif()
if(MSVC)
  target_compile_options(dsa_demo PRIVATE /W4 /utf-8)
  target_compile_options(dsa_server PRIVATE /W4 /utf-8)
else()
  target_compile_options(dsa_demo PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(dsa_server PRIVATE -Wall -Wextra -Wpedantic)
endif()
