cmake_minimum_required(VERSION 3.16)
project(DackDSA LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(dsa_core ProductLookup.cpp OrderManager.cpp)
target_include_directories(dsa_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if(MSVC)
  target_compile_options(dsa_core PRIVATE /W4 /utf-8)
else()
  target_compile_options(dsa_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(dsa_demo main.cpp)
target_link_libraries(dsa_demo PRIVATE dsa_core)
