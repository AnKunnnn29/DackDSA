cmake_minimum_required(VERSION 3.16)
project(DackDSA LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# Each executable is one translation unit; modules are included from .cpp files.
add_executable(dsa_demo main.cpp)
enable_testing()
add_executable(integration_tests tests/IntegrationTests.cpp)
target_include_directories(integration_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
add_test(NAME integration_tests COMMAND integration_tests)
add_executable(dsa_benchmark tests/Benchmark.cpp)
target_include_directories(dsa_benchmark PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
foreach(target dsa_demo integration_tests dsa_benchmark)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
